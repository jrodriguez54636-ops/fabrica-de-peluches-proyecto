#include "Material.h"
#include <iostream>

using namespace std;

/**
 * @brief Constructor de Material
 */
Material::Material(string nom, int cant, double precio) 
    : nombre(nom), cantidad(cant), precioPorUnidad(precio) {}

string Material::getNombre() const { return nombre; }
int Material::getCantidad() const { return cantidad; }
double Material::getPrecioPorUnidad() const { return precioPorUnidad; }

void Material::usarMaterial(int cantidadUsada) {
    if (cantidadUsada <= cantidad) {
        cantidad -= cantidadUsada;
    }
}

void Material::agregarStock(int cantidadAgregada) {
    cantidad += cantidadAgregada;
}
